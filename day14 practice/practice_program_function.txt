1. Calculate Total Sales

Situation: A company has sales from three different days. Create a function to calculate total sales.

def total_sales(s1, s2, s3):
    total = s1 + s2 + s3
    print("Total Sales:", total)

s1 = float(input("Enter Monday sales: "))
s2 = float(input("Enter Tuesday sales: "))
s3 = float(input("Enter Wednesday sales: "))

total_sales(s1, s2, s3)

5. Calculate Profit or Loss

Situation: A data analyst needs to determine whether a business transaction resulted in profit or loss.

def profit_loss(cost_price, selling_price):
    if selling_price > cost_price:
        profit = selling_price - cost_price
        print("Profit:", profit)
    elif cost_price > selling_price:
        loss = cost_price - selling_price
        print("Loss:", loss)
    else:
        print("No profit, no loss")

cost = float(input("Enter cost price: "))
selling = float(input("Enter selling price: "))

profit_loss(cost, selling)
